#!/bin/bash

# @raycast.schemaVersion 1
# @raycast.title [Лаб] Добавить во Входящие (Дневник)
# @raycast.mode silent
# @raycast.packageName Лабораторный полигон
# @raycast.icon 📝

export LANG="ru_RU.UTF-8"
export LC_ALL="ru_RU.UTF-8"

VAULT_DIR="/Users/getmanov/Лабораторный_полигон"
INBOX_DIR="$VAULT_DIR/Входящие"
DATE_STR=$(date "+%Y-%m-%d")
TIME_STR=$(date "+%H:%M")
TARGET="$INBOX_DIR/Дневник $DATE_STR.md"

mkdir -p "$INBOX_DIR"

CLIPBOARD_TEXT=$(pbpaste)

if [ -z "$CLIPBOARD_TEXT" ]; then
  afplay /System/Library/Sounds/Basso.aiff 2>/dev/null
  osascript -e 'display notification "Буфер обмена пуст" with title "Лабораторный полигон"'
  exit 1
fi

if [ ! -f "$TARGET" ]; then
  cat << HEADER > "$TARGET"
---
id: $(date "+%Y%m%d0000")
дата: $DATE_STR
тип: дневник
статус: сырое
теги:
  - входящие
  - дневник
---

# Дневник $DATE_STR

HEADER
fi

cat << ENTRY >> "$TARGET"

## $TIME_STR

$CLIPBOARD_TEXT
ENTRY

afplay /System/Library/Sounds/Glass.aiff 2>/dev/null
osascript -e 'display notification "Запись добавлена в дневник" with title "Лабораторный полигон"'
