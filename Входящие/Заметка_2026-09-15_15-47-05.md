# Заметка 2026-09-15_15-47-05

python3 -c '
from pathlib import Path

scripts_dir = Path(".scripts")
reference_file = scripts_dir / "clip_to_inbox.sh"

if not reference_file.exists():
    print("[!] Эталонный файл clip_to_inbox.sh не найден.")
    exit(1)

ref_content = reference_file.read_text(encoding="utf-8")
ref_lines = ref_content.splitlines()

# Извлекаем эталонные заголовки
ref_headers = {}
for line in ref_lines:
    if line.startswith("# @raycast."):
        parts = line.split(" ", 1)
        if len(parts) == 2:
            ref_headers[parts[0]] = parts[1]

print(f"Эталонные параметры: {ref_headers}")

for script in scripts_dir.iterdir():
    if script.is_file() and script.suffix in (".py", ".sh") and script.name != "clip_to_inbox.sh":
        content = script.read_text(encoding="utf-8")
        lines = content.splitlines()
        new_lines = []
        updated_headers = set()
        
        for line in lines:
            replaced = False
            for header in ref_headers:
                if line.startswith(header):
                    new_lines.append(f"{header} {ref_headers[header]}")
                    updated_headers.add(header)
                    replaced = True
                    break
            if not replaced:
                new_lines.append(line)
                
        # Добавляем недостающие заголовки сразу после шебанга
        missing = [f"{h} {ref_headers[h]}" for h in ref_headers if h not in updated_headers]
        if missing:
            if new_lines and new_lines[0].startswith("#!"):
                new_lines = [new_lines[0]] + missing + new_lines[1:]
            else:
                new_lines = missing + new_lines
                
        script.write_text("\n".join(new_lines) + "\n", encoding="utf-8")
        print(f"[+] Скрипт приведён к эталону: {script.name}")
'
