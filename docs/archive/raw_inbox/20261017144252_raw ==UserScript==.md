// ==UserScript==
// @name         Gemini: Hotkey Copy Last Response
// @namespace    http://tampermonkey.net/
// @version      1.1
// @description  Копирует последний ответ модели по Cmd+Shift+C (Mac) или Ctrl+Shift+C (Win)
// @match        https://gemini.google.com/*
// @grant        GM_setClipboard
// ==/UserScript==

(function() {
    'use strict';

    // Функция отрисовки временного индикатора подтверждения
    function showNotification(text) {
        let toast = document.getElementById('gemini-copy-toast');
        if (!toast) {
            toast = document.createElement('div');
            toast.id = 'gemini-copy-toast';
            toast.style.position = 'fixed';
            toast.style.bottom = '24px';
            toast.style.right = '24px';
            toast.style.padding = '8px 16px';
            toast.style.backgroundColor = '#1f1f1f';
            toast.style.color = '#e3e3e3';
            toast.style.borderRadius = '8px';
            toast.style.boxShadow = '0 4px 12px rgba(0,0,0,0.4)';
            toast.style.fontSize = '13px';
            toast.style.fontFamily = 'monospace';
            toast.style.zIndex = '99999';
            toast.style.transition = 'opacity 0.2s ease-in-out';
            document.body.appendChild(toast);
        }
        toast.textContent = text;
        toast.style.opacity = '1';
        setTimeout(() => { toast.style.opacity = '0'; }, 1500);
    }

    document.addEventListener('keydown', (e) => {
        // Проверяем сочетание: (Cmd ИЛИ Ctrl) + Shift + C
        const isMac = navigator.platform.toUpperCase().indexOf('MAC') >= 0;
        const metaKey = isMac ? e.metaKey : e.ctrlKey;

        if (metaKey && e.shiftKey && e.code === 'KeyC') {
            // Если курсор находится внутри поля ввода и там выделен текст — не перехватываем
            const activeTag = document.activeElement.tagName.toLowerCase();
            const isInput = activeTag === 'textarea' || activeTag === 'input' || document.activeElement.isContentEditable;
            if (isInput && window.getSelection().toString().length > 0) {
                return;
            }

            e.preventDefault();

            // Ищем контейнеры ответов ассистента в актуальном DOM Gemini
            const containers = document.querySelectorAll(
                'model-response, .model-response-text, message-content, [data-test-id="model-response-text"]'
            );

            if (!containers.length) {
                showNotification('Ответов для копирования не найдено');
                return;
            }

            // Берем самый последний сгенерированный блок
            const lastResponse = containers[containers.length - 1];
            const textToCopy = lastResponse.innerText.trim();

            if (!textToCopy) {
                showNotification('Блок ответа пуст');
                return;
            }

            // Запись в системный буфер обмена
            if (typeof GM_setClipboard !== 'undefined') {
                GM_setClipboard(textToCopy);
                showNotification('✓ Скопировано в буфер!');
            } else {
                navigator.clipboard.writeText(textToCopy)
                    .then(() => showNotification('✓ Скопировано в буфер!'))
                    .catch(() => showNotification('Ошибка доступа к буферу'));
            }
        }
    }, true);
})();
